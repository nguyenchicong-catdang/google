const SPREADSHEET_NAME = "Customer Data"; // Tên Google Sheet của bạn
const SHEET_NAME = "Sheet1"; // Tên trang tính chứa dữ liệu khách hàng

/**
 * Hàm này phục vụ trang HTML của ứng dụng web.
 * @return {GoogleAppsScript.HTML.HtmlOutput} Đối tượng HtmlOutput chứa nội dung HTML.
 */
function doGet() {
  return HtmlService.createTemplateFromFile('Index').evaluate()
    .setTitle('Quản lý Khách hàng')
    .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL);
}

/**
 * Hàm này bao gồm các chức năng phía máy chủ (backend)
 * để tương tác với Google Sheet.
 */
function include(filename) {
  return HtmlService.createHtmlOutputFromFile(filename)
    .getContent();
}

/**
 * Tìm kiếm thông tin khách hàng bằng số điện thoại.
 * @param {string} phoneNumber Số điện thoại cần tìm.
 * @return {Object[]} Mảng các đối tượng khách hàng phù hợp.
 */
function getCustomerByPhone(phoneNumber) {
  try {
    const ss = SpreadsheetApp.openByName(SPREADSHEET_NAME);
    const sheet = ss.getSheetByName(SHEET_NAME);
    if (!sheet) {
      Logger.log("Không tìm thấy trang tính: " + SHEET_NAME);
      return [];
    }

    const range = sheet.getDataRange();
    const values = range.getValues();
    const headers = values[0]; // Lấy hàng tiêu đề

    const phoneColIndex = headers.indexOf("Số điện thoại");
    if (phoneColIndex === -1) {
      Logger.log("Không tìm thấy cột 'Số điện thoại'.");
      return [];
    }

    const results = [];
    // Bắt đầu từ hàng thứ 2 để bỏ qua tiêu đề
    for (let i = 1; i < values.length; i++) {
      const row = values[i];
      // Kiểm tra nếu số điện thoại khớp (so sánh không phân biệt khoảng trắng/định dạng)
      if (String(row[phoneColIndex]).replace(/\s/g, '') === String(phoneNumber).replace(/\s/g, '')) {
        const customer = {};
        headers.forEach((header, index) => {
          customer[header] = row[index];
        });
        results.push(customer);
      }
    }
    return results;
  } catch (e) {
    Logger.log("Lỗi khi tìm kiếm khách hàng: " + e.toString());
    return { error: e.toString() };
  }
}

/**
 * Thêm một khách hàng mới vào Google Sheet.
 * @param {Object} customerData Dữ liệu khách hàng mới (Họ và Tên, Số điện thoại, Email, Địa chỉ).
 * @return {boolean|Object} true nếu thành công, hoặc đối tượng lỗi nếu có vấn đề.
 */
function addCustomer(customerData) {
  try {
    const ss = SpreadsheetApp.openByName(SPREADSHEET_NAME);
    const sheet = ss.getSheetByName(SHEET_NAME);
    if (!sheet) {
      Logger.log("Không tìm thấy trang tính: " + SHEET_NAME);
      return { error: "Không tìm thấy trang tính: " + SHEET_NAME };
    }

    // Đảm bảo có đủ cột tiêu đề
    const headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0];
    const newRow = [];
    
    // Sắp xếp dữ liệu theo thứ tự cột tiêu đề
    headers.forEach(header => {
      newRow.push(customerData[header] || ""); // Thêm giá trị hoặc chuỗi rỗng nếu không có
    });

    sheet.appendRow(newRow); // Thêm hàng mới vào cuối trang tính
    return true;
  } catch (e) {
    Logger.log("Lỗi khi thêm khách hàng: " + e.toString());
    return { error: e.toString() };
  }
}

        .withSuccessHandler(results => {
          searchSpinner.classList.add('hidden');

            return;
